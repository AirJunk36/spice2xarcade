cmake_minimum_required(VERSION 3.16)
project(spice_bio2_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SPICE_ROOT ../.. ABSOLUTE)
file(GLOB DRIVER_SOURCES "${SPICE_ROOT}/external/bi2x/*.c")
add_library(bio2_test_driver STATIC ${DRIVER_SOURCES} "${SPICE_ROOT}/bio2/device.cpp"
    "${SPICE_ROOT}/bio2/legacy.cpp" "${SPICE_ROOT}/bio2/legacy_usb.cpp")
target_include_directories(bio2_test_driver PUBLIC "${SPICE_ROOT}")
target_compile_definitions(bio2_test_driver PRIVATE BIO2_STANDALONE_TEST=1)
if(WIN32)
    target_link_libraries(bio2_test_driver PUBLIC setupapi advapi32 user32)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(bio2_test_driver PUBLIC Threads::Threads m)
endif()
add_executable(bio2_tests test_device.cpp)
target_link_libraries(bio2_tests PRIVATE bio2_test_driver)
add_test(NAME bio2_contract COMMAND bio2_tests)
add_executable(bio2_legacy_tests test_legacy.cpp)
target_link_libraries(bio2_legacy_tests PRIVATE bio2_test_driver)
add_test(NAME bio2_legacy_contract COMMAND bio2_legacy_tests)
if(NOT WIN32)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME bio2_protocol COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/run_sim.py" "$<TARGET_FILE:bio2_tests>")
    set_tests_properties(bio2_protocol PROPERTIES TIMEOUT 45)
    add_test(NAME bio2_legacy_protocol COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/run_legacy_sim.py" "$<TARGET_FILE:bio2_legacy_tests>")
    set_tests_properties(bio2_legacy_protocol PROPERTIES TIMEOUT 35)
    add_test(NAME bio2_sdvx_protocol COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/run_legacy_sim.py" "$<TARGET_FILE:bio2_legacy_tests>"
        sdvx-bi2a sdvx-kfca)
    set_tests_properties(bio2_sdvx_protocol PROPERTIES TIMEOUT 35)
endif()
